from pathlib import Path
import yaml
spec_path = Path('/home/daytona/relay-openapi.yaml')
assert spec_path.read_bytes()

paths = spec['paths']
required = {
    '/v1/chats',
    '/v1/chats/{chatId}/messages',
    '/v1/chats/{chatId}/read',
    '/v1/attachments',
    '/v1/websocket',
}
assert required <= paths.keys()
assert '/v1/events' not in paths
assert '/v1/conversations' not in paths
send = paths['/v1/chats/{chatId}/messages']['post']
assert any(p['name'] == 'Idempotency-Key' for p in send['parameters'])
assert send['requestBody']['content']['application/json']['schema']['$ref'].endswith('/SendMessageToChatRequest')
read = paths['/v1/chats/{chatId}/read']['post']
assert 'requestBody' not in read
assert read['responses']['204']['description']
websocket = paths['/v1/websocket']['get']
assert websocket['operationId'] == 'connectAgentWebSocket'
assert {'ready', 'event', 'ack', 'full_sync', 'full_sync_complete', 'ping', 'pong', 'error', 'disconnect'} == set(spec['x-relay-websocket-frames'])
schemas = spec['components']['schemas']
assert {'Chat', 'ChatHandle', 'Message', 'ParticipantAddedEvent', 'ParticipantRemovedEvent'} <= schemas.keys()
assert schemas['WebhookEnvelopeBase']['properties']['api_version']['enum'] == ['v1']
assert [str(value) for value in schemas['WebhookEnvelopeBase']['properties']['webhook_version']['enum']] == ['2026-08-30']
assert {'contact.added', 'contact.removed'} <= set(schemas['WebhookEventType']['enum'])
print('openapi_contract=pass required_paths=5 websocket_frames=9 vocabulary=Contact,Handle,Chat,Message,Participant')
